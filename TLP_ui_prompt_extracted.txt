'use client'
import { motion, useInView } from 'framer-motion'
import { useRef } from 'react'

interface Props {
  children: React.ReactNode
  delay?: number
  className?: string
}

export function AnimatedSection({ children, delay = 0, className }: Props) {
  const ref = useRef(null)
  const inView = useInView(ref, { once: true, margin: '-80px' })

  return (
    <motion.div
      ref={ref}
      initial={{ opacity: 0, y: 24 }}
      animate={inView ? { opacity: 1, y: 0 } : {}}
      transition={{ duration: 0.48, ease: [0.16, 1, 0.3, 1], delay }}
      className={className}
    >
      {children}
    </motion.div>
  )
}

10. FINAL COMMANDMENTS
'use client' on every component that uses hooks, Framer Motion, or browser APIs.
Zero hardcoded hex values in component JSX. Tailwind classes only. Inline styles only for dynamic values (classification colors, score bar gradient).
Zero border-radius except rounded-badge on classification pills.
Zero box-shadow anywhere in the codebase.
Zero Lorem Ipsum. Every string of copy is specified in this document — use it verbatim.
Zero external UI libraries except what is listed in Section 1. No shadcn components used as-is — if used, fully restyle to match this system.
AnimatedSection wraps every bento card, every section heading, every major content block. Nothing appears instantly — everything earns its place on screen.
TruthScoreWidget loops infinitely through all 3 claims with no user interaction.
All SVG icons are inline — no icon library imports. Custom-drawn stroke icons only.
next/font/google for fonts — never a <link> tag, never a CDN import.
The nav is always on top (z-50) and transitions smoothly between transparent and frosted glass states.
useInView with once: true — scroll animations never replay. Elements animate in once and stay.
Every interactive element has a hover: state. No exceptions. No static clickable elements.
transition-all duration-200 is the standard for all hover/focus transitions.
This page must pass a Lighthouse performance score of 90+. Use next/font for zero layout shift. Use loading="lazy" on any images. Keep JS bundle lean.
